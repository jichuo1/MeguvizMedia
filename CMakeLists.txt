cmake_minimum_required(VERSION 3.31)
project(MeguvizMedia VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(MGM_BUILD_TESTS "Build independent Media consumers" ON)
option(MGM_WITH_RIFE "Build optional RIFE SPI adapter" OFF)
set(MGM_COMPUTE_SOURCE "" CACHE PATH "Development Compute checkout; prefer installed package")
if(MGM_COMPUTE_SOURCE)
  set(MGC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory("${MGM_COMPUTE_SOURCE}" compute)
else()
  find_package(MeguvizCompute CONFIG REQUIRED)
endif()
find_package(Threads REQUIRED)
add_library(meguviz_media_core STATIC src/core/media.cpp src/core/timeline.cpp)
set_target_properties(meguviz_media_core PROPERTIES EXPORT_NAME core CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
add_library(MeguvizMedia::core ALIAS meguviz_media_core)
target_include_directories(meguviz_media_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(meguviz_media_core PUBLIC MeguvizCompute::core Threads::Threads)
if(MGM_WITH_RIFE)
  find_package(MeguvizRIFE CONFIG REQUIRED)
  add_library(meguviz_media_rife_integration STATIC integrations/rife/adapter.cpp)
  set_target_properties(meguviz_media_rife_integration PROPERTIES EXPORT_NAME rife_integration CXX_VISIBILITY_PRESET hidden)
  add_library(MeguvizMedia::rife_integration ALIAS meguviz_media_rife_integration)
  target_link_libraries(meguviz_media_rife_integration PUBLIC meguviz_media_core MeguvizRIFE::core)
  if(MSVC)
    target_compile_options(meguviz_media_rife_integration PRIVATE /EHsc /W4)
  endif()
  install(TARGETS meguviz_media_rife_integration EXPORT MeguvizMediaTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
if(MSVC)
  target_compile_options(meguviz_media_core PRIVATE /EHsc /W4)
else()
  target_compile_options(meguviz_media_core PRIVATE -fexceptions -Wall -Wextra -Wpedantic)
endif()
install(TARGETS meguviz_media_core EXPORT MeguvizMediaTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT MeguvizMediaTargets NAMESPACE MeguvizMedia:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeguvizMedia)
configure_package_config_file(cmake/MeguvizMediaConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/MeguvizMediaConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeguvizMedia)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/MeguvizMediaConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/MeguvizMediaConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/MeguvizMediaConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeguvizMedia)
install(FILES LICENSE NOTICE.md DESTINATION ${CMAKE_INSTALL_DATADIR}/MeguvizMedia)
add_executable(mgm_trace_replay examples/headless_timeline.cpp)
target_link_libraries(mgm_trace_replay PRIVATE MeguvizMedia::core)
if(MGM_BUILD_TESTS)
  enable_testing()
  add_executable(mgm_c_consumer tests/c_consumer.c)
  set_target_properties(mgm_c_consumer PROPERTIES LINKER_LANGUAGE CXX)
  target_link_libraries(mgm_c_consumer PRIVATE MeguvizMedia::core)
  add_test(NAME public_c_identity COMMAND mgm_c_consumer)
  add_executable(mgm_timeline_contracts tests/timeline.cpp)
  target_link_libraries(mgm_timeline_contracts PRIVATE MeguvizMedia::core)
  add_test(NAME timeline_lifecycle COMMAND mgm_timeline_contracts)
  if(WIN32)
    add_executable(mgm_abi_guard_pages tests/abi_guard_pages.cpp)
    target_link_libraries(mgm_abi_guard_pages PRIVATE MeguvizMedia::core)
    add_test(NAME abi_guard_pages COMMAND mgm_abi_guard_pages)
  endif()
  if(MGM_WITH_RIFE)
    add_executable(mgm_rife_adapter_test tests/rife_adapter.cpp)
    target_link_libraries(mgm_rife_adapter_test PRIVATE MeguvizMedia::rife_integration)
  endif()
endif()
