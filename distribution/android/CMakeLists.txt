cmake_minimum_required(VERSION 3.31)
project(MeguvizMediaIdentityJNI LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT ANDROID)
  message(FATAL_ERROR "This standalone JNI profile requires the Android toolchain")
endif()
set(MGM_IDENTITY_SDK_PREFIX "" CACHE PATH "Installed target-specific CPU/reference Media SDK")
set(MGM_IDENTITY_SDK_ROOT "" CACHE PATH "Parent directory with per-ABI installed SDKs")
if(MGM_IDENTITY_SDK_ROOT)
  set(MGM_IDENTITY_SDK_PREFIX "${MGM_IDENTITY_SDK_ROOT}/android-${ANDROID_ABI}/install")
endif()
if(NOT EXISTS "${MGM_IDENTITY_SDK_PREFIX}/lib/cmake/MeguvizMedia/MeguvizMediaConfig.cmake")
  message(FATAL_ERROR "Run tools/build_android.py first and provide its exact ABI SDK prefix")
endif()
list(PREPEND CMAKE_PREFIX_PATH "${MGM_IDENTITY_SDK_PREFIX}")
set(MeguvizCompute_DIR "${MGM_IDENTITY_SDK_PREFIX}/lib/cmake/MeguvizCompute")
find_package(MeguvizMedia CONFIG REQUIRED PATHS "${MGM_IDENTITY_SDK_PREFIX}/lib/cmake/MeguvizMedia" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
add_library(meguviz_media_identity SHARED identity_jni.cpp)
target_link_libraries(meguviz_media_identity PRIVATE MeguvizMedia::core)
target_compile_options(meguviz_media_identity PRIVATE -fexceptions -fvisibility=hidden)
target_link_options(meguviz_media_identity PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/identity.exports.map" "LINKER:-z,max-page-size=16384")
