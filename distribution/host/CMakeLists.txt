cmake_minimum_required(VERSION 3.31)
project(MeguvizCompositionFixture LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(MeguvizMedia CONFIG REQUIRED)
find_package(MeguvizRIFE CONFIG REQUIRED)
find_package(MeguvizCompute CONFIG REQUIRED)
if(NOT TARGET MeguvizMedia::rife_integration)
  message(FATAL_ERROR "The fixture requires the optional installed RIFE adapter")
endif()
add_library(meguviz_composition_fixture SHARED entry.cpp)
target_link_libraries(meguviz_composition_fixture PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,MeguvizMedia::rife_integration,MeguvizMedia::core,MeguvizRIFE::core,MeguvizCompute::core>")
set_property(TARGET meguviz_composition_fixture PROPERTY LINK_LIBRARY_OVERRIDE "WHOLE_ARCHIVE,MeguvizMedia::rife_integration,MeguvizMedia::core,MeguvizRIFE::core,MeguvizCompute::core")
set_target_properties(meguviz_composition_fixture PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
set(MGM_APPROVED_EXPORTS "" CACHE FILEPATH "Generated exact approved export list")
if(NOT EXISTS "${MGM_APPROVED_EXPORTS}")
  message(FATAL_ERROR "MGM_APPROVED_EXPORTS must name the reviewed export list")
endif()
if(MSVC)
  target_sources(meguviz_composition_fixture PRIVATE "${MGM_APPROVED_EXPORTS}")
else()
  target_link_options(meguviz_composition_fixture PRIVATE "LINKER:--version-script=${MGM_APPROVED_EXPORTS}")
endif()
